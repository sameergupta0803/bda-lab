// Task 17 - Find all books borrowed by a specific member[cite: 5]
const member = db.members.findOne({ name: "Aarav Shah" })
const borrowRecords = db.borrows.find({ member_id: member._id }).toArray()
const borrowedBookIds = borrowRecords.map(b => b.book_id)
db.books.find({ _id: { $in: borrowedBookIds } })

// Task 18 - Find overdue books[cite: 5]
db.borrows.aggregate([
  { $match: {
      $or: [
        { return_date: null },
        { return_date: { $lt: new Date() } }
      ]
  }},
  { $lookup: {
      from: "books",
      localField: "book_id",
      foreignField: "_id",
      as: "book"
  }},
  { $unwind: "$book" },
  { $project: {
      _id: 0,
      bookTitle: "$book.title",
      borrow_date: 1,
      return_date: 1,
      member_id: 1
  }}
])

// Task 19 - Count how many books each member has borrowed[cite: 5]
db.borrows.aggregate([
  { $group: {
      _id: "$member_id",
      totalBorrowed: { $sum: 1 }
  }},
  { $lookup: {
      from: "members",
      localField: "_id",
      foreignField: "_id",
      as: "member"
  }},
  { $unwind: "$member" },
  { $project: {
      _id: 0,
      memberName: "$member.name",
      totalBorrowed: 1
  }}
])

// Task 20 - Join members with the books they borrowed using $lookup[cite: 5, 6]
db.members.aggregate([
  { $lookup: {
      from: "borrows",
      localField: "_id",
      foreignField: "member_id",
      as: "borrowRecords"
  }},
  { $unwind: {
      path: "$borrowRecords",
      preserveNullAndEmptyArrays: true
  }},
  { $lookup: {
      from: "books",
      localField: "borrowRecords.book_id",
      foreignField: "_id",
      as: "bookDetails"
  }},
  { $project: {
      _id: 0,
      memberName: "$name",
      email: 1,
      bookTitle: { $arrayElemAt: ["$bookDetails.title", 0] },
      borrowDate: "$borrowRecords.borrow_date",
      returnDate: "$borrowRecords.return_date"
  }}
])
